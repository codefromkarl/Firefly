const $ = (id) => document.getElementById(id);
const MAX_BOOKMARKS = 20_000;
const MAX_BYTES = 16 * 1024 * 1024;
const UNCATEGORIZED = "未分类";
let db;
let catalog = [];

function status(message, error = false) {
	$("status").textContent = message;
	$("status").classList.toggle("error", error);
}

function openDatabase() {
	return new Promise((resolve, reject) => {
		// A separate database keeps the original Chrome import readable by the previous version.
		const request = indexedDB.open("firefly-bookmark-table", 1);
		request.onupgradeneeded = () => request.result.createObjectStore("catalog");
		request.onsuccess = () => resolve(request.result);
		request.onerror = () => reject(request.error);
		request.onblocked = () => reject(new Error("书签表被其他页面占用"));
	});
}

function store(mode, run) {
	return new Promise((resolve, reject) => {
		const tx = db.transaction("catalog", mode);
		const request = run(tx.objectStore("catalog"));
		tx.oncomplete = () => resolve(request.result);
		tx.onabort = () => reject(tx.error);
	});
}

function parse(text) {
	let data;
	try {
		data = JSON.parse(text.replace(/^\uFEFF/, ""));
	} catch {
		throw new Error("请选择整理好的 JSON 文件，不能导入 Chrome 原始 HTML。");
	}
	if (!Array.isArray(data) || !data.length || data.length > MAX_BOOKMARKS) {
		throw new Error("JSON 必须是 1 到 20,000 条书签组成的数组。");
	}
	return data.map((item, index) => {
		const title = typeof item?.title === "string" ? item.title.trim() : "";
		const url = typeof item?.url === "string" ? item.url.trim() : "";
		const purpose = typeof item?.purpose === "string" ? item.purpose.trim().replace(/\s+/gu, " ") : "";
		if (!title || !URL.canParse(url) || !purpose || title.length > 512 || url.length > 4096 || purpose.length > 240) {
			throw new Error(`第 ${index + 1} 条需要有效网址、书签名和一句用途。`);
		}
		const label = (value, field) => {
			const text = value ?? "";
			if (typeof text !== "string" || text.trim().length > 32) {
				throw new Error(`第 ${index + 1} 条的${field}须为不超过 32 字的文字。`);
			}
			return text.trim();
		};
		return { title, url, purpose, category: label(item.category, "分类"), type: label(item.type, "类型") };
	});
}

function categoryOf(bookmark) {
	return bookmark.category || UNCATEGORIZED;
}

function showCatalog(bookmarks) {
	catalog = bookmarks;
	const categories = new Set(bookmarks.map(categoryOf));
	const labels = [...categories].filter((category) => category !== UNCATEGORIZED)
		.sort(new Intl.Collator("zh-CN").compare);
	if (categories.has(UNCATEGORIZED)) labels.push(UNCATEGORIZED);
	const filter = $("category");
	filter.replaceChildren(new Option("全部分类", ""), ...labels.map((category) => new Option(category)));
	filter.hidden = categories.size < 2;
	render();
}

function render() {
	const category = $("category").value;
	const bookmarks = category ? catalog.filter((bookmark) => categoryOf(bookmark) === category) : catalog;
	const rows = document.createDocumentFragment();
	for (const { title, url, purpose, type } of bookmarks) {
		const tr = document.createElement("tr");
		const name = tr.insertCell();
		const source = new URL(url);
		if (["https:", "http:"].includes(source.protocol) && !source.username && !source.password) {
			const link = document.createElement("a");
			link.textContent = title;
			link.href = source.href;
			link.target = "_blank";
			link.rel = "noopener noreferrer";
			name.append(link);
		} else {
			name.textContent = title;
		}
		const typeCell = tr.insertCell();
		typeCell.className = "type";
		typeCell.textContent = type || "未标注";
		tr.insertCell().textContent = purpose;
		rows.append(tr);
	}
	$("items").replaceChildren(rows);
	$("count").textContent = category ? `${bookmarks.length} / ${catalog.length} 条` : `${catalog.length} 条`;
	$("count").hidden = $("table").hidden = !catalog.length;
	$("empty").hidden = !!catalog.length;
}

function showExisting(bookmarks) {
	$("existing").textContent = `当前表格有 ${bookmarks.length} 条书签。`;
	$("existing").hidden = !bookmarks.length;
}

async function importFile() {
	const fileInput = $("file");
	const button = $("import");
	const [file] = fileInput.files;
	fileInput.value = "";
	if (!file || button.disabled) return;
	button.disabled = true;
	status("");
	try {
		if (file.size > MAX_BYTES) throw new Error("文件大于 16 MB，请整理后再导入。");
		const bookmarks = parse(await file.text());
		const existing = (await store("readonly", (table) => table.get("current"))) ?? [];
		if (existing.length && !window.confirm(`用这 ${bookmarks.length} 条替换当前表格中的 ${existing.length} 条书签？`)) return;
		await store("readwrite", (table) => table.put(bookmarks, "current"));
		showExisting(bookmarks);
		status("已保存到当前浏览器。");
	} catch (error) {
		status(`未完成导入：${error?.message || "无法写入本地存储"}`, true);
	} finally {
		button.disabled = false;
	}
}

try {
	db = await openDatabase();
	const existing = (await store("readonly", (table) => table.get("current"))) ?? [];
	const importButton = $("import");
	if (importButton) {
		showExisting(existing);
		importButton.onclick = () => $("file").click();
		$("file").onchange = importFile;
		importButton.disabled = false;
	} else {
		showCatalog(existing);
		$("category").onchange = render;
	}
} catch {
	status("无法读取本地书签表，请检查浏览器的站点存储设置。", true);
}
